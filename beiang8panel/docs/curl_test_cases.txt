# BeiAng8Panel HTTP API Curl 测试用例
# 服务器地址: http://localhost:8080
# 软件版本: 1.4.2  |  协议版本: V1.22
# 用法: 复制命令到终端执行。每条命令上方 "# →" 注释为该接口的真实预期响应。
#
# 重要说明（与 docs/HTTP_API.md 一致）：
#   - 响应封装不统一：部分接口返回 {code,message,data}，部分返回裸对象/裸数组。
#   - 实现状态: ✅已实现  🔄部分实现  ⚠️模拟数据  ❌未实现(返回 code:1006 或固定假响应)
#   - /api/modbus/read 忽略 type 参数，始终按【保持寄存器(03H)】读取。
#   - 业务错误码 code:1006 表示功能未实现（HTTP 仍为 200）。
#   - 内联参数/业务错误（400/500/503）随同值 HTTP 状态码返回（body.code 与 HTTP 状态一致）；
#     下文注释"HTTP 400"即状态码 400 且 body.code=400。


# -----------------------------------------------------------------------------
# ★ Modbus 寄存器速查表（供测试人员对照；[W]=真实写 [R]=真实读 [模拟]=未真正下发）
# -----------------------------------------------------------------------------
# 保持寄存器有效范围: 0x1000-0x103E (1000H-103EH)
#
#   地址            读写   用途                 用例编号(接口)
#   1000H(4096)     [W]   通用单写演示          17/18 (/api/modbus/write)
#   1000H(4096)     [R]   通用读演示(03H)       14/15/16 (/api/modbus/read)
#   1001H           [W]   新风模块开关          6/7 (/api/freshair/switch; 单写, 见下方互斥说明)
#   1001H           [R]   新风状态缓存之一      9 (/api/freshair/status)
#   1002H           [W]   超净模式开关          14/15 (/api/super-pure/switch; 单写)
#   1002H           [R]   超净状态缓存          17 (/api/super-pure/status)
#   1003H           [W]   调湿模块开关          10/11 (/api/humidity-module/switch; 单写)
#   1003H           [R]   调湿状态缓存之一      13 (/api/humidity-module/status)
#   ※ 模块互斥联动(v1.22, 由4CP设备端自动执行, 屏只下发单条开关命令):
#     开新风→设备自动关超净; 开超净→设备自动关新风+调湿; 开调湿→设备自动关超净;
#     新风与调湿不互斥可共存。联动生效需数秒(写后读回刷新), 验证序列见用例17b。
#   1006H           [R]   一键离家开关(v1.22)   周期采集缓存(暂无对外接口)
#   1007H           [W]   新风运行模式          22 (/api/freshair/runmode; 前置1001H=1)
#   1007H           [R]   风速只读前置判定      18/21 (=3自动/5睡眠时1008H拒写)
#   1008H           [W]   新风风速档位          18 (/api/freshair/speed; 前置1001H=1且
#                                                               1007H∈{0,1,2,4}; 上限: 模式2→2003H,
#                                                               模式0/1/4→2004H, 不满足均HTTP 400)
#   100FH           [W]   目标湿度              26 (/api/humidity-module/target)
#   101BH-101FH     [W]   RTC时间(10H批量5寄存器)  33/34 (/api/rtc/time; 1016H-101AH预留)
#   2000H-203BH     [R]   输入寄存器周期采集块  6/9 (04H整块60个; 环境数据/风量最大档等由此解析)
#   2003H/2004H     [R]   风量上限+状态缓存    9/18 (status返回; 1008H写校验上限)
#   200FH/2013H     [R]   室内/室外PM2.5缓存    8 (/api/idle/air-quality-reminder)
#   1004H/1005H     [W]   加湿/除湿开关          72/73 (前置1003H=0; 见用例17b后互斥说明)
#   1009H           [W]   排风风量档位(预留)     74 (/api/freshair/exhaust-speed)
#   100AH           [W]   整机运行模式           76 (/api/device/unit-run-mode)
#   100BH-100EH     [W]   无极风量/占空比        77/78 (/api/freshair/stepless,/duty)
#   1010H-1015H     [W]   温度/消毒/辅热/强度等  79-83 (§15组: ×10存储见79/83)
#   1020H           [W]   延时关风机             84 (/api/device/fan-delay-off)
#   1028H-1040H     [W]   压缩机/厂测/滤网复位   85-91 (1040H恢复出厂高危需confirm)
#   1041H-1076H     [W]   厂测标定(🔒厂测锁)     92/93 (未进厂测模式→400)
#   1016H           [模拟] 批量写演示(未下发)   19 (/api/modbus/write-multiple)
#   离散0/32        [模拟] 读离散输入(未调02H)  20 (/api/modbus/read-discrete)

# =============================================================================
# 一、设备控制 API
# =============================================================================

# 1. 获取设备状态  ✅ ▸寄存器: R 纯缓存接口(本调用不触发 Modbus)——数据来自采集线程
#    周期读: 保持寄存器整块 1000H-1076H(03H,119个) + 输入寄存器 2000H-203BH(04H,60个) + 离散 3000H+0-76(02H,77位);
#    本接口消费字段: 1000H总开关/1001H新风/1002H超净/1003H调湿/1004H加湿/1005H除湿/1006H一键离家/
#    1007H运行模式/1008H风量档位/100AH整机模式/100BH无极风量/1011H等离子/1012H IEF/
#    2000H工厂标志/2001H机型/2002H版本/2003H·2004H最大档位/201CH电辅热/201DH自动循环显示
# → {"code":0,"message":"Success","data":{"device":{"factoryFlag":"BA","deviceModel":"4CP","version":"0101","deviceAddress":209},"control":{"switchOn":false,"runMode":0,"controlMode":0,"iefPurification":false,"fanGear":0,"fanMaxGear":6,"auxHeat":0,"humidificationOn":false,"dehumidificationOn":false}}}
curl "http://localhost:8080/api/device/status"

# 2. 快捷开机  ✅ ▸寄存器: W 1001H=1 (单写; 若超净开着由设备端自动联动关闭, 新风与调湿可共存)
# → {"accepted":true,"power":true}（裸对象，无 code 外层）
curl -X POST -H "Content-Type: application/json" -d '{"power":true}' "http://localhost:8080/api/device/power"

# 3. 快捷关机  ✅ ▸寄存器: W 1001H-1003H=0,0,0 (10H组写一帧; 关闭不触发设备联动, 需全关)
# → {"accepted":true,"power":false}
curl -X POST -H "Content-Type: application/json" -d '{"power":false}' "http://localhost:8080/api/device/power"

# 4. 切换新风运行模式（mode=0~5, v1.22）  ✅ ▸寄存器: W 1007H (前置 R 1001H=1, 否则 HTTP 400)
# → {"code":0,"message":"success","data":{"accepted":true,"mode":2}}
curl -X POST -H "Content-Type: application/json" -d '{"mode":2}' "http://localhost:8080/api/device/mode"

# 5. 遍历所有运行模式  ✅ ▸寄存器: W 1007H×6 (0内循环/1内循环·混风/2全热新风·节能新风/3自动/
#    4旁通·换气/5睡眠; 前置1001H=1; 连续写建议间隔≥3秒等读回缓存刷新, 否则可能HTTP 400)
for mode in 0 1 2 3 4 5; do
  curl -s -X POST -H "Content-Type: application/json" -d "{\"mode\":$mode}" "http://localhost:8080/api/device/mode"; echo
done

# 6. 新风模块开启  ✅ ▸寄存器: W 1001H=1 (单写; 若超净开着设备自动关超净)
# → {"code":0,"message":"success","data":{"accepted":true,"on":true}}
curl -X POST -H "Content-Type: application/json" -d '{"on":true}' "http://localhost:8080/api/freshair/switch"

# 7. 新风模块关闭  ✅ ▸寄存器: W 1001H=0
# → {"code":0,"message":"success","data":{"accepted":true,"on":false}}
curl -X POST -H "Content-Type: application/json" -d '{"on":false}' "http://localhost:8080/api/freshair/switch"

# 8. 新风开关参数校验（on 缺失应返回 HTTP 400）
# → {"code":400,"message":"参数 on（布尔）必填"}
curl -X POST -H "Content-Type: application/json" -d '{}' "http://localhost:8080/api/freshair/switch"

# 9. 查询新风状态  ✅ ▸寄存器: R 1001H新风开关/1007H运行模式/1008H风量档位/100CH新风占空比/
#    2003H新风最大档/2004H内循环最大档 (纯缓存接口; v1.3.3 起新增 100CH, 移除原 1006H/201DH 字段)
# → {"freshAirModuleOn":true,"runMode":2,"fanGear":3,"freshFanDutyCycle":65,"fanMaxGear":3,"fanMaxGearRecirc":5}
curl "http://localhost:8080/api/freshair/status"

# 10. 调湿模块开启  ✅ ▸寄存器: W 1003H=1 (单写; 若超净开着设备自动关超净)
# → {"code":0,"message":"success","data":{"accepted":true,"on":true}}
curl -X POST -H "Content-Type: application/json" -d '{"on":true}' "http://localhost:8080/api/humidity-module/switch"

# 11. 调湿模块关闭  ✅ ▸寄存器: W 1003H=0
# → {"code":0,"message":"success","data":{"accepted":true,"on":false}}
curl -X POST -H "Content-Type: application/json" -d '{"on":false}' "http://localhost:8080/api/humidity-module/switch"

# 12. 调湿开关参数校验（on 缺失应返回 HTTP 400）
# → {"code":400,"message":"参数 on（布尔）必填"}
curl -X POST -H "Content-Type: application/json" -d '{}' "http://localhost:8080/api/humidity-module/switch"

# 13. 查询调湿模块状态  ✅ ▸寄存器: R 1003H/100FH(缓存)
# → {"humidityModuleOn":true,"targetHumidity":50}
curl "http://localhost:8080/api/humidity-module/status"

# 14. 超净模式开启  ✅ ▸寄存器: W 1002H=1 (单写; 设备自动关新风/调湿, 联动验证见用例17b)
# → {"code":0,"message":"success","data":{"accepted":true,"on":true}}
curl -X POST -H "Content-Type: application/json" -d '{"on":true}' "http://localhost:8080/api/super-pure/switch"

# 15. 超净模式关闭  ✅ ▸寄存器: W 1002H=0
# → {"code":0,"message":"success","data":{"accepted":true,"on":false}}
curl -X POST -H "Content-Type: application/json" -d '{"on":false}' "http://localhost:8080/api/super-pure/switch"

# 16. 超净开关参数校验（on 缺失应返回 HTTP 400）
# → {"code":400,"message":"参数 on（布尔）必填"}
curl -X POST -H "Content-Type: application/json" -d '{}' "http://localhost:8080/api/super-pure/switch"

# 17. 查询超净模式状态  ✅ ▸寄存器: R 1002H(缓存)
# → {"superPureOn":true}
curl "http://localhost:8080/api/super-pure/status"

# 17b. 模块互斥联动验证  ✅ ▸寄存器: W 1002H=1 → R 1001H/1002H/1003H (联动由4CP设备端自动执行,
#      屏只下发单条命令; 联动生效需数秒, 每步间 sleep 8)
curl -X POST -H "Content-Type: application/json" -d '{"on":true}' "http://localhost:8080/api/freshair/switch"; sleep 8
curl -X POST -H "Content-Type: application/json" -d '{"on":true}' "http://localhost:8080/api/super-pure/switch"; sleep 8
curl "http://localhost:8080/api/freshair/status"    # → freshAirModuleOn:false (新风被设备联动关闭)
curl "http://localhost:8080/api/super-pure/status"  # → {"superPureOn":true}
curl "http://localhost:8080/api/modbus/read?address=0x1001&count=3"  # → values:[0,1,0]

# 18. 新风风速设定（档位3）  ✅ ▸寄存器: W 1008H (前置: R 1001H=1 且 1007H∈{0,1,2,4}, 3自动/
#     5睡眠只读; 档位上限: 模式2受2003H(4CP=3), 模式0/1/4受2004H(≤5); 任一不满足均 HTTP 400)
# → {"code":0,"message":"success","data":{"accepted":true,"speed":3}}
curl -X POST -H "Content-Type: application/json" -d '{"speed":3}' "http://localhost:8080/api/freshair/speed"

# 19. 风速参数校验（speed 缺失应返回 HTTP 400）
# → {"code":400,"message":"参数 speed（整数）必填"}
curl -X POST -H "Content-Type: application/json" -d '{}' "http://localhost:8080/api/freshair/speed"

# 20. 风速越界校验（speed=9 应返回 HTTP 400）
# → {"code":400,"message":"风量档位必须在0-6之间"}
curl -X POST -H "Content-Type: application/json" -d '{"speed":9}' "http://localhost:8080/api/freshair/speed"

# 21. 自动/睡眠模式下设定风速  ▸寄存器: 前置 R 1007H=3(自动)或5(睡眠) → 拒写 1008H (HTTP 400 只读)
# → {"code":400,"message":"当前运行模式(1007H=3自动/5睡眠)下风量档位只读"}
curl -X POST -H "Content-Type: application/json" -d '{"speed":2}' "http://localhost:8080/api/freshair/speed"

# 22. 新风运行模式设置（全热新风 mode=2）  ✅ ▸寄存器: W 1007H (前置 R 1001H=1, 否则 HTTP 400)

# 25. 新风关闭时设置运行模式  ▸寄存器: 前置 R 1001H=0 → 拒写 1007H (HTTP 400 只读)

# 6. 获取室内/室外/送风环境数据  ✅ ▸寄存器: R 200DH-201AH 共14个(接口纯读缓存; 数据来自
#    周期采集 04H 输入寄存器整块 2000H-203BH, 本接口只消费其中 200DH-201AH)
#    indoorReturnAir (RA1 室内回风): 200DH 温度(有符号÷10,℃) / 200EH 湿度(%) / 200FH PM2.5 / 2010H CO2(ppm)
#    outdoorAir      (OA 室外新风):   2011H 温度 / 2012H 湿度 / 2013H PM2.5 / 2014H CO2
#    supplyAir       (SA 送风):       2015H 温度 / 2016H 湿度 / 2017H PM2.5 / 2018H CO2
#    airQuality:                     2019H TVOC / 201AH 甲醛 (均原值÷100, mg/m³)
# → {"code":200,"message":"Success","data":{"indoorReturnAir":{...},"outdoorAir":{...},"supplyAir":{...},"airQuality":{"tvoc":...,"formaldehyde":...}}}
curl "http://localhost:8080/api/idle/environment"

# 7. 用 jq 提取关键环境字段
curl -s "http://localhost:8080/api/idle/environment" | jq '.data | .indoorReturnAir, .outdoorAir, .airQuality'

# 8. 获取室内外空气评价提醒（待机页面显示）  ✅ ▸寄存器: R 200FH(室内PM2.5)/2013H(室外PM2.5)(缓存)
# → {"code":200,"message":"Success","data":{"indoorPM25":<int>,"outdoorPM25":<int>,"indoorLevel":"优|良|中|差","outdoorLevel":"优|良|中|差","reminder":"<文案或空串>"}}
#    reminder 依室内外等级组合而定（共7种情形；室外为优/良时为空串）；分级 优0-35/良36-75/中76-150/差151-999。详见 docs/HTTP_API.md
curl "http://localhost:8080/api/idle/air-quality-reminder"

# 9. 用 jq 提取提醒文案与室内外等级
curl -s "http://localhost:8080/api/idle/air-quality-reminder" | jq '.data | {reminder, indoorLevel, outdoorLevel, indoorPM25, outdoorPM25}'

# 34. 设置 RTC 时间  ✅ ▸寄存器: W 101BH-101FH 共5个(10H批量写; 1016H-101AH预留, v1.22布局)

#     请求体支持 "time"（"YYYY-M-D H:M:S"，无前导零亦可）或 "timestamp"；
#     可选 "format"：0=24小时制(默认) / 1=12小时制(由前端设置, 时间值仍按24h传入,
#     12h打包(时1-12+bit7=PM)由后端换算; GET 在12h制下 hour 归一为24h并附 meridiem)
#     严格日历校验: 非法日期(如 2026-2-30、平年2-29)返回 code=1 且不写寄存器

curl -X POST -H "Content-Type: application/json" -d '{"timestamp":1722900000}' "http://localhost:8080/api/rtc/time"
curl -X POST -H "Content-Type: application/json" -d '{"time":"2026-9-3 13:05:07","format":1}' "http://localhost:8080/api/rtc/time"  # → 12h: 101DH=0x8105(1PM), 101FH=0x0100
curl -X POST -H "Content-Type: application/json" -d '{"time":"2026-2-30 10:00:00"}' "http://localhost:8080/api/rtc/time"            # → code:1 非法日期, 不写寄存器
